82 [[noreturn]]
83 void report_no_matches(CallReport call, std::vector<MatchReport> noMatchReports) override
84 {
85 auto const msg = stringify_no_matches(call, noMatchReports);
86 if (m_Out)
87 {
88 *m_Out << msg << '\n';
89 }
90
91 throw UnmatchedCallT{msg, std::move(call), call.fromLoc};
92 }
93
94 [[noreturn]]
95 void report_inapplicable_matches(CallReport call, std::vector<MatchReport> reports) override
96
97 {
98 const auto msg = stringify_inapplicable_matches(call, reports);
99 if (m_Out)
100 {
101 *m_Out << msg << '\n';
102 }
103
104 throw UnmatchedCallT{msg, std::move(call), call.fromLoc};
105 }
106
107 void report_full_match(
108 [[maybe_unused]] CallReport const call,
109 [[maybe_unused]] MatchReport const match) noexcept override
110 {
111 MIMICPP_ASSERT(std::holds_alternative<state_applicable>(match.expectationReport.controlReport), "Report denotes inapplicable expectation.");
112 }
113
114 void report_unfulfilled_expectation(ExpectationReport expectationReport) override
115 {
116 if (0 == std::uncaught_exceptions())
117 {
118 auto const msg = stringify_unfulfilled_expectation(expectationReport);
119 if (m_Out)
120 {
121 *m_Out << msg << '\n';
122 }
123
124 throw UnfulfilledExpectationT{msg, std::move(expectationReport)};
125 }
126 }
127
128 void report_error(const StringT message) override
129 {
130 if (0 == std::uncaught_exceptions())
131 {
132 if (m_Out)
133 {
134 *m_Out << message << '\n';
135 }
136
137 throw Error{message};
138 }
139 }
140
141 void report_unhandled_exception(
142 const CallReport call,
143 const ExpectationReport expectationReport,
144 const std::exception_ptr exception) override
145 {
146 if (m_Out)
147 {
148 *m_Out << stringify_unhandled_exception(call, expectationReport, exception)
149 << '\n';
150 }
151 }
152
153 private:
154 std::ostream* m_Out{};
155 };
