30 template <typename Matcher>
31 struct matcher_matches_fn
32 {
33 using matches_fn = mimicpp::detail::matches_hook::matches_fn;
34
35 public:
36 Matcher const& matcher;
37
38 template <typename... Args>
39 requires std::invocable<matches_fn, Matcher const&, Args&...>
40 [[nodiscard]]
41 // projected arguments may come as value, so Args& won't work in all cases
42 // just forward them as lvalue-ref
43 MatchResult operator()(Args&&... args) const
44 noexcept(std::is_nothrow_invocable_v<matches_fn, Matcher const&, Args&...>)
45 {
46 return mimicpp::detail::matches_hook::matches(matcher, args...);
47 }
48 };

50 template <typename Matcher, typename MatchesStrategy, typename DescribeStrategy>
51 class ArgsRequirement
52 {
53 public:
54 [[nodiscard]]
55 explicit constexpr ArgsRequirement(Matcher matcher, MatchesStrategy matchesStrategy, DescribeStrategy describeStrategy)
56 noexcept(
57 std::is_nothrow_move_constructible_v<Matcher>
58 && std::is_nothrow_move_constructible_v<MatchesStrategy>
59 && std::is_nothrow_move_constructible_v<DescribeStrategy>)
60 : m_Matcher{std::move(matcher)},
61 m_MatchesStrategy{std::move(matchesStrategy)},
62 m_DescribeStrategy{std::move(describeStrategy)}
63 {
64 }
65
66 [[nodiscard]]
67 static constexpr bool is_satisfied() noexcept
68 {
69 return true;
70 }
71
72 template <typename Return, typename... Args>
73 requires std::is_invocable_r_v<MatchResult, MatchesStrategy const&, matcher_matches_fn<Matcher>, call::Info<Return, Args...> const&>
74 [[nodiscard]]
75 MatchResult matches(call::Info<Return, Args...> const& info) const
76 noexcept(std::is_nothrow_invocable_v<MatchesStrategy const&, matcher_matches_fn<Matcher>, call::Info<Return, Args...> const&>)
77 {
78 matcher_matches_fn<Matcher> const matches{m_Matcher};
79 auto result = std::invoke(m_MatchesStrategy, matches, info);
80 std::visit([&](auto& inner) {
81 if (auto& description = inner.description)
82 {
83 description = std::invoke(m_DescribeStrategy, *description);
84 }
85 },
86 result);
87
88 return result;
89 }
90
91 template <typename Return, typename... Args>
92 static constexpr void consume([[maybe_unused]] call::Info<Return, Args...> const& info) noexcept
93 {
94 }
95
96 private:
97 Matcher m_Matcher;
98 [[no_unique_address]] MatchesStrategy m_MatchesStrategy;
99 [[no_unique_address]] DescribeStrategy m_DescribeStrategy;
100 };

130 template <
131 std::size_t... indices,
132 typename Matcher,
133 typename... Projections>
134 [[nodiscard]]
135 constexpr auto make_args_policy(
136 Matcher&& matcher,
137 std::tuple<Projections...>&& projections)
138 {
139 static_assert(
140 sizeof...(indices) == sizeof...(Projections),
141 "Indices and projections size mismatch.");
142
143 using arg_selector_t = args_selector_fn<
144 std::add_lvalue_reference_t,
145 std::index_sequence<indices...>>;
146 using apply_strategy_t = arg_list_indirect_apply_fn<std::remove_cvref_t<Projections>...>;
147 using describe_strategy_t = arg_requirement_describer<indices...>;
148
149 return ArgsRequirement{
150 std::forward<Matcher>(matcher),
151 apply_args_fn(
152 arg_selector_t{},
153 apply_strategy_t{std::move(projections)}),
154 describe_strategy_t{}};
155 }
156}

219 template <

221 std::size_t... others,
222 typename Matcher,
223 typename... Projections>
224 [[nodiscard]]
225 constexpr auto args(Matcher&& matcher, Projections&&... projections)
226 noexcept(
227 std::is_nothrow_constructible_v<std::remove_cvref_t<Matcher>, Matcher>
228 && (... && std::is_nothrow_move_constructible_v<std::remove_cvref_t<Projections>>))
229 {
230 static_assert(
231 sizeof...(projections) <= 1u + sizeof...(others),
232 "The projection count exceeds the amount of indices.");
233
234 return expectation::policies::detail::make_args_policy<first, others...>(
235 std::forward<Matcher>(matcher),
236 util::detail::expand_tuple<std::identity, 1u + sizeof...(others)>(
237 std::forward_as_tuple(std::forward<Projections>(projections)...)));
