100 [[nodiscard]]
101 constexpr bool apply(CVQualifier const qualifier) noexcept
102 {
103 switch (qualifier)
104 {
105 case CVQualifier::id_const:
106 return !std::exchange(isConst, true);

108 case CVQualifier::id_volatile:

151 struct OperatorFunctionId
152 {
153 using Symbol = std::variant<
154 lexing::operator_or_punctuator,
155 std::pair<lexing::keyword, bool /*isArray*/>,
156 std::array<lexing::operator_or_punctuator, 2u>>;
157 Symbol symbol;
158
159 [[nodiscard]]
160 constexpr bool is_call() const noexcept
161 {
162 if (auto const* const doubleOp = std::get_if<std::array<lexing::operator_or_punctuator, 2u>>(&symbol))
163 {
164 return lexing::operator_or_punctuator{"("} == doubleOp->front()
165 && lexing::operator_or_punctuator{")"} == doubleOp->back();
166 }
167
168 return false;
169 }

171 [[nodiscard]]
172 bool operator==(OperatorFunctionId const&) const = default;

317 [[nodiscard]]
318 MIMICPP_DETAIL_CONSTEXPR_PRETTY_TYPES bool try_apply(lexing::keyword const& keyword)
319 {
320 return try_apply_size_spec(keyword)
321 || try_apply_signed_spec(keyword)
322 || try_apply_base(keyword);
323 }
324
325 [[nodiscard]]
326 bool operator==(BuiltinType const&) const = default;
327
328 private:
329 [[nodiscard]]
330 constexpr bool try_apply_base(lexing::keyword const& keyword) noexcept
331 {
332 // Todo: verify correct type keywords
333 if (!base)
334 {
335 base = keyword;
336 return true;
337 }
338
339 return false;
340 }
341
342 [[nodiscard]]
343 MIMICPP_DETAIL_CONSTEXPR_PRETTY_TYPES bool try_apply_signed_spec(lexing::keyword const& keyword) noexcept
344 {
345 std::optional const spec = std::invoke([&]() -> std::optional<SignedSpec> {
346 if (lexing::keyword{"unsigned"} == keyword)
347 {
348 return SignedSpec::id_unsigned;
349 }
350
351 if (lexing::keyword{"signed"} == keyword)
352 {
353 return SignedSpec::id_signed;
354 }
355
356 return std::nullopt;
357 });
358
359 if (spec
360 && !signedSpec)
361 {
362 signedSpec = spec;
363 return true;
364 }
365
366 return false;
367 }
368
369 [[nodiscard]]
370 MIMICPP_DETAIL_CONSTEXPR_PRETTY_TYPES bool try_apply_size_spec(lexing::keyword const& keyword) noexcept
371 {
372 std::optional const spec = std::invoke([&]() -> std::optional<SizeSpec> {
373 if (lexing::keyword{"long"} == keyword)
374 {
375 return SizeSpec::id_long;
376 }
377
378 if (lexing::keyword{"short"} == keyword)
379 {
380 return SizeSpec::id_short;
381 }
382
383 // This is a msvc symbol
384 if (lexing::keyword{"__int64"} == keyword)
385 {
386 return SizeSpec::id_longlong;
387 }
388
389 return std::nullopt;
390 });
391
392 if (spec)
393 {
394 if (!sizeSpec)
395 {
396 sizeSpec = spec;
397 return true;
398 }
399
400 if (*sizeSpec == SizeSpec::id_long
401 && *spec == SizeSpec::id_long)
402 {
403 sizeSpec = SizeSpec::id_longlong;
404 return true;
405 }
406 }
407
408 return false;
409 }
