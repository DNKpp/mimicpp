219 struct spec_info
220 {
221 bool hasConst{false};
222 ValueCategory refQualifier = ValueCategory::any;
223 bool hasNoexcept{false};
224 bool hasOverride{false};
225 bool hasFinal{false};
226 bool isValid{true};
227 };
228
229 [[nodiscard]]
230 constexpr spec_info evaluate_specs(std::string_view specs)
231 {
232 constexpr std::string_view spaces{" \t"};
233 constexpr std::string_view letters{"abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"};
234
235 spec_info info{};
236 auto const set_once = [&](bool& flag) {
237 info.isValid = info.isValid && !std::exchange(flag, true);
238 };
239 auto const set_ref = [&](ValueCategory const category) {
240 info.isValid = info.isValid && ValueCategory::any == std::exchange(info.refQualifier, category);
241 };
242 auto const skip_spaces = [&] {
243 specs.remove_prefix(std::min(specs.find_first_not_of(spaces), specs.size()));
244 };
245
246 for (skip_spaces();
247 info.isValid && !specs.empty();
248 skip_spaces())
249 {
250 if (specs.starts_with("&&"))
251 {
252 set_ref(ValueCategory::rvalue);
253 specs.remove_prefix(2u);
254 }
255 else if (specs.starts_with('&'))
256 {
257 set_ref(ValueCategory::lvalue);
258 specs.remove_prefix(1u);
259 }
260 else
261 {
262 auto const length = std::min(specs.find_first_not_of(letters), specs.size());
263 std::string_view const keyword = specs.substr(0u, length);
264 specs.remove_prefix(length);
265 if ("const" == keyword)
266 {
267 set_once(info.hasConst);
268 }
269 else if ("noexcept" == keyword)
270 {
271 set_once(info.hasNoexcept);
272 }
273 else if ("override" == keyword)
274 {
275 set_once(info.hasOverride);
276 }
277 else if ("final" == keyword)
278 {
279 set_once(info.hasFinal);
280 }
281 else
282 {
283 info.isValid = false;
284 }
285 }
286 }
287
288 return info;
289 }
290
291 [[nodiscard]]
292 constexpr bool is_supported_spec(std::string_view const specs)
293 {
294 return evaluate_specs(specs).isValid;
295 }

297 template <auto specText>
298 struct apply_normalized_specs
299 {
300 static constexpr auto info = evaluate_specs(specText.view());
301 static_assert(info.isValid, "Unsupported specifier.");
302
303 template <typename Signature>
304 [[nodiscard]]
305 static consteval auto evaluate() noexcept
306 {
307 using sig_maybe_ref = std::conditional_t<
308 ValueCategory::lvalue == info.refQualifier,
309 signature_add_lvalue_ref_qualifier_t<Signature>,
310 std::conditional_t<
311 ValueCategory::rvalue == info.refQualifier,
312 signature_add_rvalue_ref_qualifier_t<Signature>,
313 Signature>>;
314
315 using sig_maybe_const = std::conditional_t<
316 info.hasConst,
317 signature_add_const_qualifier_t<sig_maybe_ref>,
318 sig_maybe_ref>;
319
320 using sig_maybe_noexcept = std::conditional_t<
321 info.hasNoexcept,
322 signature_add_noexcept_t<sig_maybe_const>,
323 sig_maybe_const>;
324
325 return std::type_identity<sig_maybe_noexcept>{};
326 }
327
328 template <typename Signature>
329 using type = decltype(evaluate<Signature>())::type;
330 };
331
332 template <typename RawSignature, auto specText>
333 using apply_normalized_specs_t = apply_normalized_specs<specText>::template type<RawSignature>;
334}
