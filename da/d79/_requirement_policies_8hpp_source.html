31 template <typename Matcher>
32 struct matcher_matches_fn
33 {
34 using matches_fn = mimicpp::detail::matches_hook::matches_fn;
35
36 public:
37 Matcher const& matcher;
38
39 template <typename... Args>
40 requires std::invocable<matches_fn, Matcher const&, Args&...>
41 [[nodiscard]]
42 // projected arguments may come as value, so Args& won't work in all cases
43 // just forward them as lvalue-ref
44 MatchResult operator()(Args&&... args) const
45 noexcept(std::is_nothrow_invocable_v<matches_fn, Matcher const&, Args&...>)
46 {
47 return mimicpp::detail::matches_hook::matches(matcher, args...);
48 }

51 template <typename Matcher, typename MatchesStrategy, typename DescribeStrategy>
52 class ArgsRequirement
53 {
54 public:
55 [[nodiscard]]
56 explicit constexpr ArgsRequirement(Matcher matcher, MatchesStrategy matchesStrategy, DescribeStrategy describeStrategy)
57 noexcept(
58 std::is_nothrow_move_constructible_v<Matcher>
59 && std::is_nothrow_move_constructible_v<MatchesStrategy>
60 && std::is_nothrow_move_constructible_v<DescribeStrategy>)
61 : m_Matcher{std::move(matcher)},
62 m_MatchesStrategy{std::move(matchesStrategy)},
63 m_DescribeStrategy{std::move(describeStrategy)}
64 {
65 }
66
67 [[nodiscard]]
68 static constexpr bool is_satisfied() noexcept
69 {
70 return true;
71 }
72
73 template <typename Return, typename... Args>
74 requires std::is_invocable_r_v<MatchResult, MatchesStrategy const&, matcher_matches_fn<Matcher>, call::Info<Return, Args...> const&>
75 [[nodiscard]]
76 MatchResult matches(call::Info<Return, Args...> const& info) const
77 noexcept(std::is_nothrow_invocable_v<MatchesStrategy const&, matcher_matches_fn<Matcher>, call::Info<Return, Args...> const&>)
78 {
79 matcher_matches_fn<Matcher> const matches{m_Matcher};
80 auto result = std::invoke(m_MatchesStrategy, matches, info);
81 std::visit(
82 [&](auto& inner) {
83 if (auto& description = inner.description)
84 {
85 description = std::invoke(m_DescribeStrategy, *description);
86 }
87 },
88 result);
89
90 return result;
91 }
92
93 template <typename Return, typename... Args>
94 static constexpr void consume([[maybe_unused]] call::Info<Return, Args...> const& info) noexcept
95 {
96 }
97
98 private:
99 Matcher m_Matcher;
100 [[no_unique_address]] MatchesStrategy m_MatchesStrategy;
101 [[no_unique_address]] DescribeStrategy m_DescribeStrategy;
102 };

154 {

156 StringT operator()(StringViewT const matcherDescription) const

177 template <
178 std::size_t... indices,
179 typename Matcher,
180 typename... Projections>
181 [[nodiscard]]
182 constexpr auto make_args_policy(
183 Matcher&& matcher,
184 std::tuple<Projections...>&& projections)
185 {
186 static_assert(
187 sizeof...(indices) == sizeof...(Projections),
188 "Indices and projections size mismatch.");
189
190 using arg_selector_t = args_selector_fn<
191 std::add_lvalue_reference_t,
192 std::index_sequence<indices...>>;
193 using apply_strategy_t = arg_list_indirect_apply_fn<std::remove_cvref_t<Projections>...>;
194 using describe_strategy_t = arg_requirement_describer<indices...>;
195
196 return ArgsRequirement{
197 std::forward<Matcher>(matcher),
198 apply_args_fn(
199 arg_selector_t{},
200 apply_strategy_t{std::move(projections)}),
201 describe_strategy_t{}};
202 }
203}
