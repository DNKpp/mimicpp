129 template <print_iterator OutIter>
130 constexpr OutIter prettify_type(OutIter out, std::string_view const name)
131 {
132 if (std::optional const typeId = parse_type(name))
133 {
134 parsing::PrintVisitor<OutIter> visitor{std::move(out)};
135 visitor.visit(*typeId);
136
137 return visitor.out();
138 }
139
140 return format::format_to(std::move(out), "{}", name);
141 }
