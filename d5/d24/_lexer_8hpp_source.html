27 // This is a constexpr friendly version of `std::isspace`.
28 // see: https://en.cppreference.com/w/cpp/string/byte/isspace
29 inline auto constexpr is_space = [](char const c) noexcept {
30 switch (c)
31 {
32 case ' ': [[fallthrough]];
33 case '\f': [[fallthrough]];
34 case '\n': [[fallthrough]];
35 case '\r': [[fallthrough]];
36 case '\t': [[fallthrough]];
37 case '\v': return true;
38
39 default: return false;
40 }
41 };

261 [[nodiscard]]
262 constexpr token find_next() noexcept
263 {
264 if (m_Text.empty())
265 {
266 return token{
267 .content = {m_Text.cend(), m_Text.cend()},
268 .classification = end{}
269 };
270 }
271
272 if (is_space(m_Text.front()))
273 {
274 // Multiple consecutive spaces or any whitespace character other than a single space
275 // carry no meaningful semantic value beyond delimitation.
276 // Although single spaces may sometimes influence the result and sometimes not,
277 // complicating the overall process, we filter out all non-single whitespace characters here.
278 if (StringViewT const content = next_as_space();
279 " " == content)
280 {
281 return token{
282 .content = content,
283 .classification = space{}};
284 }
285
286 return find_next();
287 }
288
289 if (auto const options = util::prefix_range(
290 operator_or_punctuator::textCollection,
291 m_Text.substr(0u, 1u)))
292 {
293 return next_as_op_or_punctuator(options);
294 }
295
296 if (std::optional literal = try_next_as_literal())
297 {
298 return token{
299 .content = literal->content,
300 .classification = *std::move(literal)};
301 }
302
303 StringViewT const content = next_as_identifier();
304 // As we do not perform any prefix-checks, we need to check now whether the token actually denotes a keyword.
305 if (auto const iter = util::binary_find(keyword::textCollection, content);
306 iter != keyword::textCollection.cend())
307 {
308 return token{
309 .content = content,
310 .classification = keyword{std::ranges::distance(keyword::textCollection.begin(), iter)}};
311 }
312
313 return token{
314 .content = content,
315 .classification = identifier{.content = content}};
316 }

318 [[nodiscard]]
319 constexpr StringViewT next_as_space() noexcept
320 {
321 auto const end = std::ranges::find_if_not(m_Text.cbegin() + 1, m_Text.cend(), is_space);
322 StringViewT const content{m_Text.cbegin(), end};
323 m_Text = StringViewT{end, m_Text.cend()};
324
325 return content;
326 }

332 [[nodiscard]]
333 constexpr token next_as_op_or_punctuator(std::span<StringViewT const> options) noexcept
334 {
335 MIMICPP_ASSERT(m_Text.substr(0u, 1u) == options.front(), "Assumption does not hold.");
336
337 auto const try_advance = [&, this](std::size_t const n) {
338 if (n <= m_Text.size())
339 {
340 return util::prefix_range(
341 options,
342 StringViewT{m_Text.cbegin(), m_Text.cbegin() + n});
343 }
344
345 return std::ranges::subrange{options.end(), options.end()};
346 };
347
348 std::size_t length{1u};
349 StringViewT const* lastMatch = &options.front();
350 while (auto const nextOptions = try_advance(length + 1))
351 {
352 ++length;
353 options = {nextOptions.begin(), nextOptions.end()};
354
355 // If the first string is exactly the size of the prefix, it's a match.
356 if (auto const& front = options.front();
357 length == front.size())
358 {
359 lastMatch = &front;
360 }
361 }
362
363 MIMICPP_ASSERT(!options.empty(), "Invalid state.");
364 MIMICPP_ASSERT(lastMatch, "Invalid state.");
365
366 auto const index = std::ranges::distance(operator_or_punctuator::textCollection.data(), lastMatch);
367 StringViewT const content{m_Text.substr(0u, lastMatch->size())};
368 m_Text.remove_prefix(lastMatch->size());
369
370 return token{
371 .content = content,
372 .classification = operator_or_punctuator{index}};
373 }
374
385 [[nodiscard]]
386 constexpr StringViewT next_as_identifier() noexcept
387 {
388 auto const last = std::ranges::find_if_not(
389 m_Text.cbegin() + 1,
390 m_Text.cend(),
391 [](auto const c) {
392 return !is_space(c)
393 && !std::ranges::binary_search(operator_or_punctuator::textCollection, StringViewT{&c, 1u});
394 });
395
396 StringViewT const content{m_Text.cbegin(), last};
397 m_Text = {last, m_Text.cend()};
398
399 return content;
400 }

402 // This function currently implements just a very basic integer-literal detection.
403 // see: https://eel.is/c++draft/lex.literal.kinds#nt:literal
404 [[nodiscard]]
405 constexpr std::optional<literal> try_next_as_literal() noexcept
406 {
407 MIMICPP_ASSERT(!m_Text.empty(), "Empty text.");
408
409 if (is_digit(m_Text.front()))
410 {
411 if ('0' == m_Text.front()
412 && 2u <= m_Text.size())
413 {
414 // Todo: one of
415 // https://eel.is/c++draft/lex.icon#nt:binary-literal
416 // https://eel.is/c++draft/lex.icon#nt:octal-literal
417 // https://eel.is/c++draft/lex.icon#nt:hexadecimal-prefix
418 }
419
420 // Todo: this is just a very naive approach.
421 auto const last = std::ranges::find_if_not(m_Text.cbegin() + 1, m_Text.cend(), is_digit);
422 StringViewT const content{m_Text.cbegin(), last};
423 m_Text = {last, m_Text.cend()};
424
425 return literal{.content = content};
426 }
427
428 return std::nullopt;
429 }
